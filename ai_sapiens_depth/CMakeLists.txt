cmake_minimum_required(VERSION 3.16)
project(ai_sapiens_depth)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(ai_sapiens_interfaces REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(${PROJECT_NAME} SHARED src/depth_processor.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} sensor_msgs ai_sapiens_interfaces)
target_link_libraries(${PROJECT_NAME} yaml-cpp)
add_executable(depth_history_node src/depth_history_node.cpp)
ament_target_dependencies(depth_history_node rclcpp sensor_msgs ai_sapiens_interfaces)
target_link_libraries(depth_history_node ${PROJECT_NAME})
install(TARGETS depth_history_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_depth_processor test/test_depth_processor.cpp)
  target_link_libraries(test_depth_processor ${PROJECT_NAME})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(sensor_msgs ai_sapiens_interfaces yaml-cpp)
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_package()
